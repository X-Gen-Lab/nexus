foreach(component EVENT STATE_MACHINE)
    if(CONFIG_FRAMEWORK_${component})
        message(FATAL_ERROR "Framework ${component} has no maintained implementation")
    endif()
endforeach()
foreach(component log shell config init)
    string(TOUPPER "${component}" symbol)
    if(CONFIG_FRAMEWORK_${symbol})
        add_subdirectory(${component})
    elseif(NEXUS_BUILD_TESTS)
        message(FATAL_ERROR "Existing framework tests require CONFIG_FRAMEWORK_${symbol}")
    endif()
endforeach()
