# Selected memory sources and private dependencies.
nexus_controller_modes(stm32_external_bus_modes external_bus)
if(stm32_external_bus_modes)
    set(stm32_gpio_helpers ON)
endif()
if("qspi-indirect-sdr" IN_LIST stm32_external_bus_modes OR
        "qspi-indirect-dma" IN_LIST stm32_external_bus_modes)
    target_sources(nexus_soc_stm32f4 PRIVATE
        "${CMAKE_CURRENT_LIST_DIR}/qspi.c" "${CMAKE_CURRENT_LIST_DIR}/w25q128jv.c")
    nexus_stm32f4_qspi_hal(nexus_soc_stm32f4)
endif()
if("qspi-indirect-dma" IN_LIST stm32_external_bus_modes)
    target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/qspi_dma.c")
    target_link_libraries(nexus_soc_stm32f4 PRIVATE Nexus::IODMA)
    target_compile_definitions(nexus_soc_stm32f4 PRIVATE NEXUS_QSPI_PROGRAM_DMA=1)
endif()
if("sdram-controller" IN_LIST stm32_external_bus_modes)
    target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fmc_sdram.c")
    nexus_stm32f4_sdram_ll(nexus_soc_stm32f4)
endif()
if("async16" IN_LIST stm32_external_bus_modes OR "external-memory" IN_LIST stm32_external_bus_modes)
    target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fsmc.c")
    nexus_stm32f407_fsmc_ll(nexus_soc_stm32f4)
    if("flash_async" IN_LIST NEXUS_SELECTED_KINDS)
        target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fsmc_nor.c")
    endif()
endif()
if("nand" IN_LIST NEXUS_SELECTED_KINDS)
    target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fsmc_nand.c")
endif()
if(NEXUS_QSPI_CPU_MAP)
    target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/qspi_mapped.c")
    target_link_libraries(nexus_soc_stm32f4 PUBLIC Nexus::MappedStorage PRIVATE Nexus::IOFlashAsync)
    target_compile_definitions(nexus_soc_stm32f4 PRIVATE NEXUS_QSPI_CPU_MAP=1)
endif()
if(NEXUS_QSPI_QUAD_READ)
    target_compile_definitions(nexus_soc_stm32f4 PRIVATE NEXUS_QSPI_QUAD_READ=1)
endif()
if(NEXUS_SDRAM_GRAPHICS)
    target_compile_definitions(nexus_soc_stm32f4 PRIVATE NX_STM32_SDRAM_GRAPHICS=1)
endif()
