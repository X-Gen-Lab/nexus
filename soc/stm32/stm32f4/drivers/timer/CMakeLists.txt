# Selected timer sources and private dependencies.
nexus_controller_modes(pwm_modes pwm)
if(pwm_modes)
    set(stm32_gpio_helpers ON)
    target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/timer.c")
endif()
nexus_controller_modes(encoder_modes encoder)
if(encoder_modes)
    set(stm32_gpio_helpers ON)
    target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/encoder.c")
endif()
nexus_controller_names(advanced_timers advanced_pwm)
foreach(timer IN LISTS advanced_timers)
    set(stm32_gpio_helpers ON)
    if(timer STREQUAL "TIM1")
        target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/advanced_pwm.c")
    elseif(timer STREQUAL "TIM8")
        target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/advanced_pwm_tim8.c")
    else()
        message(FATAL_ERROR "Unreviewed advanced timer: ${timer}")
    endif()
endforeach()
nexus_controller_names(capture_timers timer_capture)
foreach(timer IN LISTS capture_timers)
    set(stm32_gpio_helpers ON)
    if(timer STREQUAL "TIM4")
        target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/timer_capture.c")
    elseif(timer MATCHES "^TIM(1|3|5|8|9|1[0-4])$")
        string(TOLOWER "${timer}" timer_source_name)
        target_sources(nexus_soc_stm32f4 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/timer_capture_${timer_source_name}.c")
    else()
        message(FATAL_ERROR "Unreviewed capture timer: ${timer}")
    endif()
endforeach()
foreach(kind pwm_group timer_pulse timer_sequence)
    nexus_controller_modes(waveform_modes "${kind}")
    if(waveform_modes)
        set(stm32_gpio_helpers ON)
        target_sources(nexus_soc_stm32f4 PRIVATE
            "${CMAKE_CURRENT_LIST_DIR}/timer_waveform.c" "${CMAKE_CURRENT_LIST_DIR}/../ip/av/av.c")
        if(NOT kind STREQUAL "pwm_group")
            string(TOUPPER "${kind}" waveform_definition)
            target_compile_definitions(nexus_soc_stm32f4 PRIVATE "NEXUS_STM32_${waveform_definition}=1")
        endif()
    endif()
endforeach()
