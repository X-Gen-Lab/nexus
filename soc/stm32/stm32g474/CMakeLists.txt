# Reviewed G474 controller modes; official CMSIS inputs remain private.
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/platform/CMSISCore.cmake")
set(g4_device "${NEXUS_SOURCE_DIR}/vendors/st/cmsis_device_g4")
nexus_cmsis_core_include(g4_cmsis)
set(g4_lock "${NEXUS_SOURCE_DIR}/dependencies/stm32g4.lock.json")
set(g4_verifier "${NEXUS_SOURCE_DIR}/tools/maintenance/stm32g4_lock.py")
execute_process(
    COMMAND "${Python3_EXECUTABLE}" "${g4_verifier}"
        --source "${g4_device}" --lock "${g4_lock}"
    RESULT_VARIABLE g4_lock_result
    OUTPUT_VARIABLE g4_lock_output
    ERROR_VARIABLE g4_lock_error)
if(NOT g4_lock_result EQUAL 0)
    message(FATAL_ERROR "G474 source lock rejected: ${g4_lock_error}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${g4_lock}" "${g4_verifier}" "${g4_device}/LICENSE.md"
    "${g4_device}/Include/stm32g474xx.h"
    "${g4_device}/Include/stm32g4xx.h"
    "${g4_device}/Include/system_stm32g4xx.h"
    "${g4_device}/Source/Templates/gcc/startup_stm32g474xx.s")
include("${NEXUS_SOURCE_DIR}/cmake/platform/ControllerModes.cmake")
set(g4_sources)
nexus_controller_modes(g4_gpio_modes gpio)
nexus_controller_modes(g4_uart_modes uart)
nexus_controller_modes(g4_exti_modes exti)
nexus_controller_modes(g4_watchdog_modes watchdog)
nexus_controller_modes(g4_spi_modes spi)
nexus_controller_modes(g4_pwm_modes pwm)
nexus_controller_modes(g4_pwm_group_modes pwm_group)
nexus_controller_modes(g4_capture_modes timer_capture)
nexus_controller_modes(g4_encoder_modes encoder)
nexus_controller_modes(g4_adc_modes adc)
nexus_controller_modes(g4_adc_group_modes adc_group)
nexus_controller_modes(g4_i2c_modes i2c)
nexus_controller_modes(g4_flash_modes flash)
nexus_controller_modes(g4_pwm_adc_modes pwm_adc)
nexus_controller_modes(g4_can_fd_modes can_fd)
if(g4_gpio_modes OR g4_uart_modes OR g4_spi_modes OR g4_pwm_modes
        OR g4_pwm_group_modes OR g4_capture_modes OR g4_encoder_modes
        OR g4_pwm_adc_modes OR g4_can_fd_modes OR g4_adc_group_modes)
    list(APPEND g4_sources drivers/gpio.c)
endif()
if("named-states" IN_LIST g4_gpio_modes)
    list(APPEND g4_sources drivers/pin_state.c)
endif()
if(g4_uart_modes)
    list(APPEND g4_sources drivers/uart.c)
endif()
if(g4_exti_modes)
    list(APPEND g4_sources drivers/exti.c)
endif()
if(g4_watchdog_modes)
    list(APPEND g4_sources drivers/watchdog.c)
endif()
if(g4_spi_modes)
    list(APPEND g4_sources drivers/spi.c)
endif()
if("fixed-pwm" IN_LIST g4_pwm_modes)
    list(APPEND g4_sources drivers/pwm.c)
endif()
if("shared-fixed-pwm" IN_LIST g4_pwm_modes)
    list(APPEND g4_sources drivers/pwm_shared.c)
endif()
if(g4_pwm_group_modes OR g4_pwm_adc_modes)
    list(APPEND g4_sources drivers/pwm_group.c)
endif()
if(g4_capture_modes)
    list(APPEND g4_sources drivers/timer_capture.c)
endif()
if(g4_encoder_modes)
    list(APPEND g4_sources drivers/encoder.c)
endif()
if(g4_adc_modes OR g4_pwm_adc_modes)
    list(APPEND g4_sources drivers/adc.c)
    if("trigger-dma" IN_LIST g4_adc_modes)
        list(APPEND g4_sources drivers/adc_stream.c)
    endif()
endif()
if(g4_pwm_adc_modes)
    list(APPEND g4_sources drivers/pwm_adc.c)
endif()
if(g4_adc_group_modes)
    list(APPEND g4_sources drivers/adc_group.c)
endif()
if(g4_can_fd_modes)
    list(APPEND g4_sources drivers/fdcan.c)
endif()
if(g4_i2c_modes)
    list(APPEND g4_sources drivers/i2c.c "${NEXUS_SOURCE_DIR}/drivers/i2c/i2c_v2/i2c_v2.c")
endif()
if(g4_flash_modes)
    list(APPEND g4_sources drivers/flash.c)
endif()
if(g4_sources)
    add_library(nexus_soc_stm32g474 STATIC ${g4_sources})
    target_link_libraries(nexus_soc_stm32g474 PUBLIC nexus_io
        PRIVATE Nexus::Arch nexus_build_options nexus_owned_warnings)
    target_include_directories(nexus_soc_stm32g474 PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/private"
        "${NEXUS_SOURCE_DIR}/drivers/i2c/i2c_v2"
        "${NEXUS_SOURCE_DIR}/soc/stm32/common"
        "${NEXUS_SOURCE_DIR}/io/private")
    if(g4_adc_modes OR g4_pwm_adc_modes OR g4_can_fd_modes OR g4_adc_group_modes)
        include("${NEXUS_SOURCE_DIR}/cmake/platform/STM32Official.cmake")
        nexus_stm32_official(nexus_soc_stm32g474 g4)
    endif()
    target_include_directories(nexus_soc_stm32g474 SYSTEM PRIVATE
        "${g4_device}/Include")
    nexus_use_cmsis_core(nexus_soc_stm32g474)
    target_compile_definitions(nexus_soc_stm32g474 PRIVATE STM32G474xx)
    if(g4_can_fd_modes)
        target_compile_definitions(nexus_soc_stm32g474 PRIVATE
            USE_RTOS=0 USE_HAL_FDCAN_REGISTER_CALLBACKS=0)
    endif()
else()
    add_library(nexus_soc_stm32g474 INTERFACE)
    target_link_libraries(nexus_soc_stm32g474 INTERFACE nexus_io)
endif()
set(NEXUS_STARTUP_SOURCE
    "${g4_device}/Source/Templates/gcc/startup_stm32g474xx.s" PARENT_SCOPE)
set(NEXUS_SYSTEM_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/system.c" PARENT_SCOPE)
set(g4_binding_private "${CMAKE_CURRENT_SOURCE_DIR}/private")
if(g4_i2c_modes)
    list(APPEND g4_binding_private "${NEXUS_SOURCE_DIR}/drivers/i2c/i2c_v2")
endif()
if(g4_adc_group_modes)
    list(APPEND g4_binding_private "${NEXUS_SOURCE_DIR}/soc/stm32/common")
endif()
set(NEXUS_SYSTEM_PRIVATE_INCLUDES ${g4_binding_private} PARENT_SCOPE)
set(NEXUS_SYSTEM_VENDOR_INCLUDES "${g4_device}/Include" "${g4_cmsis}"
    PARENT_SCOPE)
set(NEXUS_SYSTEM_COMPILE_DEFINITIONS STM32G474xx PARENT_SCOPE)
