add_library(nexus_arch_interface INTERFACE)
target_include_directories(nexus_arch_interface INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(nexus_arch_interface INTERFACE nexus_build_options)

if(NEXUS_PLATFORM STREQUAL "native")
    set(nexus_arch_port native)
elseif(NEXUS_PLATFORM MATCHES "^(stm32|gd32f470)$" AND CONFIG_CPU_ARCH STREQUAL "cortex-m4")
    set(nexus_arch_port cortex_m4)
else()
    message(FATAL_ERROR "No maintained Nexus architecture port for ${NEXUS_PLATFORM}/${CONFIG_CPU_ARCH}")
endif()
add_library(nexus_arch STATIC "${nexus_arch_port}/nx_arch_${nexus_arch_port}.c")
target_link_libraries(nexus_arch PUBLIC nexus_arch_interface)
if(nexus_arch_port STREQUAL "native" AND NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(nexus_arch PRIVATE Threads::Threads)
endif()
add_library(Nexus::Arch ALIAS nexus_arch)
add_library(Nexus::ArchInterface ALIAS nexus_arch_interface)
