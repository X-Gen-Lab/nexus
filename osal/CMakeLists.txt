add_library(osal_interface INTERFACE)
target_include_directories(osal_interface INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(osal_interface INTERFACE nexus_build_options)
if(NOT NEXUS_OSAL_BACKEND MATCHES "^(native|baremetal|freertos)$")
    message(FATAL_ERROR "OSAL backend '${NEXUS_OSAL_BACKEND}' is not implemented")
endif()
add_library(osal STATIC "adapters/${NEXUS_OSAL_BACKEND}/osal_${NEXUS_OSAL_BACKEND}.c")
target_link_libraries(osal PUBLIC osal_interface)
if(NEXUS_OSAL_BACKEND STREQUAL "native")
    find_package(Threads REQUIRED)
    target_link_libraries(osal PUBLIC Threads::Threads)
elseif(NEXUS_OSAL_BACKEND STREQUAL "freertos")
    if(NOT NEXUS_PLATFORM STREQUAL "stm32")
        message(FATAL_ERROR "FreeRTOS reference port currently requires STM32F407")
    endif()
    add_library(freertos_config INTERFACE)
    target_include_directories(freertos_config INTERFACE "${CMAKE_SOURCE_DIR}/osal/adapters/freertos")
    target_link_libraries(freertos_config INTERFACE nexus_build_options)
    set(FREERTOS_PORT GCC_ARM_CM4F CACHE STRING "Pinned reference port" FORCE)
    set(FREERTOS_HEAP 4 CACHE STRING "Heap implementation" FORCE)
    add_subdirectory("${CMAKE_SOURCE_DIR}/ext/freertos" "${CMAKE_BINARY_DIR}/ext/freertos")
    target_link_libraries(osal PUBLIC freertos_kernel freertos_config)
endif()
