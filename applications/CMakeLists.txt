# Generic examples and target-specific references use the same target helper.
foreach(application blinky shell_demo config_demo stm32_blinky stm32_config_test freertos_demo)
    string(TOUPPER "${application}" application_symbol)
    if(CONFIG_BUILD_APP_${application_symbol})
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${application}/CMakeLists.txt")
            message(FATAL_ERROR "Enabled application ${application} has no implementation")
        endif()
        add_subdirectory(${application})
    endif()
endforeach()
