28/* Task context only. The caller serializes feed/poll/recover. No allocations.
29 * read returns a coherent snapshot; write_atomic either commits EVERY value or
30 * changes NONE. authorize is required and applies to broadcasts as well.
31 * Critical writes check product authorization/interlocks here and recheck them
32 * atomically inside write_atomic if product state has concurrent owners. RTU
33 * itself authenticates neither a peer nor a request. All callbacks are bounded.
34 */

36 void* context;
37 nx_modbus_register_result_t (*read)(void*, uint16_t, uint16_t, uint16_t*);

42} nx_modbus_registers_t;

44/* All deadlines are absolute monotonic microseconds. write may use DMA, but
45 * drain must observe the final stop bit before success. After any write/drain
46 * error, abort_and_settle MUST stop DMA and settle all callbacks before
47 * success. If abort fails the link is latched FAULTED, its TX storage is
48 * retained, and DE remains asserted until recover settles the port. This is a
49 * port failure, not a normal timeout. A port must never retain bytes after
50 * successful drain or abort. direction(false) must leave the transceiver in
51 * receive mode.
52 */
53typedef struct {
54 void* context;
55 uint64_t (*now_us)(void*);
56 bool (*direction)(void*, bool transmit);
57 bool (*write)(void*, const uint8_t*, size_t, uint64_t deadline_us);
58 bool (*drain)(void*, uint64_t deadline_us);
59 bool (*abort_and_settle)(void*, uint64_t deadline_us);
60} nx_modbus_rs485_port_t;
