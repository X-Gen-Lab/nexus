# Usage requirements belong to targets, never directory-global compiler flags.
add_library(nexus_build_options INTERFACE)
string(TOUPPER "${NEXUS_PLATFORM}" nexus_platform_symbol)
target_compile_definitions(nexus_build_options INTERFACE "NEXUS_PLATFORM_${nexus_platform_symbol}" NEXUS_EFFECTIVE_CONFIG=1)
file(SHA256 "${NEXUS_CONFIG_DIR}/effective.config" nexus_config_sha256)
set(nexus_source_revision "unknown-source")
if(EXISTS "${NEXUS_SOURCE_DIR}/.nexus-source-sdk.json")
    execute_process(COMMAND "${Python3_EXECUTABLE}"
        "${NEXUS_SOURCE_DIR}/cmake/package/package_source_sdk.py" --verify "${NEXUS_SOURCE_DIR}"
        RESULT_VARIABLE _nexus_sdk_verification ERROR_VARIABLE _nexus_sdk_diagnostic)
    if(NOT _nexus_sdk_verification EQUAL 0)
        message(FATAL_ERROR "Source SDK identity verification failed: ${_nexus_sdk_diagnostic}")
    endif()
    file(READ "${NEXUS_SOURCE_DIR}/.nexus-source-sdk.json" _nexus_sdk_manifest)
    string(JSON nexus_source_revision GET "${_nexus_sdk_manifest}" source_revision)
    string(JSON _nexus_sdk_publishable GET "${_nexus_sdk_manifest}" publishable)
    if(NOT _nexus_sdk_publishable)
        if(NOT NEXUS_ALLOW_SOURCE_SDK_FIXTURE)
            message(FATAL_ERROR "Development source SDK fixture requires NEXUS_ALLOW_SOURCE_SDK_FIXTURE=ON")
        endif()
        string(JSON _nexus_snapshot_identity GET "${_nexus_sdk_manifest}" snapshot_sha256)
        string(SUBSTRING "${_nexus_snapshot_identity}" 0 12 _nexus_snapshot_short)
        string(APPEND nexus_source_revision "-snapshot-${_nexus_snapshot_short}")
    endif()
else()
    find_package(Git QUIET)
    if(Git_FOUND)
        # An archived source folder inside a consumer Git checkout is not the
        # consumer commit. Only this checkout's own root can establish Git identity.
        execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --show-toplevel
            WORKING_DIRECTORY "${NEXUS_SOURCE_DIR}" RESULT_VARIABLE _nexus_git_root_result
            OUTPUT_VARIABLE _nexus_git_root OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        file(REAL_PATH "${NEXUS_SOURCE_DIR}" _nexus_own_root)
        if(_nexus_git_root_result EQUAL 0)
            file(REAL_PATH "${_nexus_git_root}" _nexus_actual_git_root)
        endif()
        if(_nexus_git_root_result EQUAL 0 AND _nexus_actual_git_root STREQUAL _nexus_own_root)
            execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --verify HEAD
                WORKING_DIRECTORY "${NEXUS_SOURCE_DIR}" RESULT_VARIABLE revision_result
                OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
            if(revision_result EQUAL 0)
                set(nexus_source_revision "${revision}")
                execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain --untracked-files=normal -- .
                    WORKING_DIRECTORY "${NEXUS_SOURCE_DIR}" RESULT_VARIABLE status_result
                    OUTPUT_VARIABLE source_changes ERROR_QUIET)
                if(NOT status_result EQUAL 0 OR source_changes)
                    string(APPEND nexus_source_revision "-dirty")
                endif()
            endif()
        endif()
    endif()
endif()
target_compile_definitions(nexus_build_options INTERFACE
    "NEXUS_BOARD_SHA256=\"${NEXUS_BOARD_SHA256}\""
    "NEXUS_LAYOUT_SHA256=\"${NEXUS_LAYOUT_SHA256}\""
    "NEXUS_CONFIG_SHA256=\"${nexus_config_sha256}\""
    "NEXUS_SOURCE_REVISION=\"${nexus_source_revision}\"")

    OSAL_MAX_QUEUE_ITEM_SIZE=${CONFIG_OSAL_MAX_QUEUE_ITEM_SIZE}
    OSAL_MAX_QUEUE_BYTES=${CONFIG_OSAL_MAX_QUEUE_BYTES})
foreach(resource TASKS MUTEXES SEMS QUEUES EVENTS TIMERS)
    target_compile_definitions(nexus_build_options INTERFACE
        OSAL_MAX_${resource}=${CONFIG_OSAL_MAX_${resource}})
endforeach()
target_include_directories(nexus_build_options INTERFACE "${NEXUS_CONFIG_DIR}")
target_compile_features(nexus_build_options INTERFACE c_std_11)
set_target_properties(nexus_build_options PROPERTIES
    NEXUS_SOURCE_DIR "${NEXUS_SOURCE_DIR}"
    NEXUS_BINARY_DIR "${NEXUS_BINARY_DIR}"
    NEXUS_CONFIG_DIR "${NEXUS_CONFIG_DIR}"
    NEXUS_CONFIG_CMAKE "${NEXUS_CONFIG_DIR}/config.cmake"
    NEXUS_PLATFORM "${NEXUS_PLATFORM}"
    NEXUS_PLATFORM_TARGET "platform_${NEXUS_PLATFORM}")
add_library(Nexus::Config ALIAS nexus_build_options)
if(MSVC)
    target_compile_options(nexus_build_options INTERFACE /W4 /FS)
else()
    target_compile_options(nexus_build_options INTERFACE -Wall -Wextra -ffunction-sections -fdata-sections)
endif()
if(NEXUS_PLATFORM MATCHES "^(stm32|gd32f470)$")
    if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_C_COMPILER MATCHES "arm-none-eabi")
        message(FATAL_ERROR "${NEXUS_PLATFORM} requires the ARM GCC toolchain preset")
    endif()
    configure_file("${NEXUS_SOURCE_DIR}/dependencies/toolchains.lock.json"
        "${NEXUS_CONFIG_DIR}/toolchains.lock.json" COPYONLY)
    set(nexus_cpu_options -mcpu=${CONFIG_CPU_ARCH} -mthumb -mfloat-abi=${CONFIG_FLOAT_ABI})
    if(CONFIG_FPU_TYPE)
        list(APPEND nexus_cpu_options -mfpu=${CONFIG_FPU_TYPE})
    endif()
    target_compile_options(nexus_build_options INTERFACE ${nexus_cpu_options})
    target_link_options(nexus_build_options INTERFACE ${nexus_cpu_options}
        --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections)
elseif(NOT NEXUS_PLATFORM STREQUAL "native")
    message(FATAL_ERROR "Platform '${NEXUS_PLATFORM}' has no maintained driver implementation")
endif()
if(NEXUS_ENABLE_COVERAGE)
    if(NOT NEXUS_PLATFORM STREQUAL "native" OR NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "Coverage requires the native GCC configuration")
    endif()
    target_compile_options(nexus_build_options INTERFACE --coverage -fprofile-update=atomic -O0 -g)
    target_link_options(nexus_build_options INTERFACE --coverage)
endif()
if(NEXUS_ENABLE_SANITIZERS)
    if(NOT NEXUS_PLATFORM STREQUAL "native" OR NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizers require the native GCC or Clang configuration")
    endif()
    target_compile_options(nexus_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(nexus_build_options INTERFACE -fsanitize=address,undefined)
endif()
