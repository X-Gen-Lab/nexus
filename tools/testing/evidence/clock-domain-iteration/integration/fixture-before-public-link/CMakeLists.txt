cmake_minimum_required(VERSION 3.31)
project(sleep_consumer LANGUAGES C ASM)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(NEXUS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NEXUS_BUILD_WORKLOAD OFF CACHE BOOL "" FORCE)
if(NOT IS_DIRECTORY "${NEXUS_SLEEP_SOURCE}")
    message(FATAL_ERROR "Set the Nexus source directory for this consumer")
endif()
add_subdirectory("${NEXUS_SLEEP_SOURCE}" nexus-sdk)
option(SLEEP_SELECTED "Explicit caller-owned sleep mechanism" OFF)
get_target_property(assembly_family Nexus::Platform NEXUS_SOC_FAMILY)
get_target_property(sdk_root Nexus::Platform NEXUS_SOURCE_DIR)
if(SLEEP_SELECTED)
    if(assembly_family STREQUAL "stm32f407")
        nexus_add_stm32f407_sleep(selected_sleep)
    elseif(assembly_family STREQUAL "gd32f470")
        nexus_add_gd32f470_sleep(selected_sleep)
    else()
        message(FATAL_ERROR "Unsupported sleep fixture SoC")
    endif()
    nexus_add_firmware(sleep_consumer SOURCES main.c LIBRARIES selected_sleep)
    target_include_directories(sleep_consumer PRIVATE
        "${sdk_root}/components/clock_domain/private"
        "${sdk_root}/soc/${assembly_family}/private")
    target_compile_definitions(sleep_consumer PRIVATE SLEEP_SELECTED=1
        "SLEEP_${assembly_family}=1")
else()
    nexus_add_firmware(sleep_consumer SOURCES main.c)
endif()
