foreach(peripheral I2C TIMER ADC DAC CAN USB ETHERNET SDIO)
    if(CONFIG_STM32_${peripheral}_ENABLE)
        message(FATAL_ERROR "Nexus STM32 ${peripheral} driver has no maintained implementation")
    endif()
endforeach()
# A supported build is a concrete SoC and toolchain combination.
if(NOT CONFIG_STM32F407 OR NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "The maintained STM32 reference build is STM32F407 with ARM GCC")
endif()
if(NOT CONFIG_CPU_ARCH STREQUAL "cortex-m4")
    message(FATAL_ERROR "STM32F407 requires Cortex-M4 architecture")
endif()
set(cmsis "${CMAKE_SOURCE_DIR}/vendors/arm/CMSIS_5/CMSIS/Core/Include")
set(device "${CMAKE_SOURCE_DIR}/vendors/st/cmsis_device_f4")
set(vendor "${CMAKE_SOURCE_DIR}/vendors/st/stm32f4xx_hal_driver")
foreach(required "${cmsis}/core_cm4.h" "${device}/Include/stm32f407xx.h" "${vendor}/Src/stm32f4xx_hal.c")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing pinned submodule file ${required}; initialize dependencies")
    endif()
endforeach()
add_library(cmsis_device INTERFACE)
target_include_directories(cmsis_device INTERFACE "${cmsis}" "${device}/Include")
target_compile_definitions(cmsis_device INTERFACE STM32F407xx STM32F4 USE_HAL_DRIVER)
target_link_libraries(cmsis_device INTERFACE nexus_build_options)
# Vendor SDK sources remain isolated from project warning policy.
file(GLOB vendor_sources CONFIGURE_DEPENDS "${vendor}/Src/*.c")
list(FILTER vendor_sources EXCLUDE REGEX "_template\\.c$")
add_library(hal_driver STATIC ${vendor_sources})
target_include_directories(hal_driver PUBLIC "${vendor}/Inc" "${vendor}/Inc/Legacy" "${CMAKE_CURRENT_SOURCE_DIR}/config")
target_link_libraries(hal_driver PUBLIC cmsis_device)
target_compile_options(hal_driver PRIVATE -Wno-unused-parameter)

set(platform_sources
    src/boot/stm32_platform_init.c
    src/clock/stm32_clock_common.c src/clock/stm32f4_clock.c
    src/interrupt/stm32_interrupt.c src/interrupt/stm32_isr_manager.c src/interrupt/stm32f4_it.c
    src/platform/nx_platform_assert.c
    src/system/stm32_error.c src/system/stm32_error_handler.c src/system/stm32_chip_validation.c
    src/system/stm32_performance.c src/system/stm32_profiler.c
    "${device}/Source/Templates/system_stm32f4xx.c")
foreach(peripheral gpio uart spi)
    string(TOUPPER "${peripheral}" symbol)
    if(CONFIG_STM32_${symbol}_ENABLE)
        file(GLOB driver_sources CONFIGURE_DEPENDS "src/hal/${peripheral}/*.c")
        list(APPEND platform_sources ${driver_sources})
    endif()
endforeach()
add_library(platform_stm32 OBJECT ${platform_sources}
    "${CMAKE_SOURCE_DIR}/soc/stm32f407vg/flash.c")
target_include_directories(platform_stm32 PUBLIC "${CMAKE_SOURCE_DIR}/soc/stm32f407vg")
target_link_libraries(platform_stm32 PUBLIC nexus_storage)
if(CONFIG_BOARD_STM32F4DISCOVERY)
    target_sources(platform_stm32 PRIVATE "${CMAKE_SOURCE_DIR}/boards/stm32f4discovery/identity.c")
    target_include_directories(platform_stm32 PUBLIC "${CMAKE_SOURCE_DIR}/boards/stm32f4discovery")
endif()
if(CONFIG_BOARD_STM32F4DISCOVERY AND CONFIG_STM32_SPI_ENABLE)
    target_sources(platform_stm32 PRIVATE "${CMAKE_SOURCE_DIR}/boards/stm32f4discovery/spi.c")
    target_include_directories(platform_stm32 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/hal/spi")
endif()
target_include_directories(platform_stm32 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/config")
target_compile_definitions(platform_stm32 PUBLIC NEXUS_PLATFORM_STM32 HSE_VALUE=${CONFIG_STM32_HSE_VALUE})
target_link_libraries(platform_stm32 PUBLIC hal osal hal_driver)
set(NEXUS_STARTUP_FILE "${device}/Source/Templates/gcc/startup_stm32f407xx.s" PARENT_SCOPE)
set(NEXUS_LINKER_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/linker/stm32f4/gcc/stm32f407.ld" PARENT_SCOPE)
