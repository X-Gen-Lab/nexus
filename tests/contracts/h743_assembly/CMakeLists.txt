cmake_minimum_required(VERSION 3.31)
project(h743_assembly_link_fixture LANGUAGES C ASM)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT NEXUS_SOURCE_ROOT OR NOT NEXUS_ASSEMBLY_FILE)
    message(FATAL_ERROR "H743 fixture needs source root and one explicit assembly")
endif()
set(NEXUS_BUILD_TESTS OFF CACHE BOOL "No host tests in ARM fixture")
set(NEXUS_BUILD_WORKLOAD OFF CACHE BOOL "External explicit fixture main")
add_subdirectory("${NEXUS_SOURCE_ROOT}" nexus-sdk)
nexus_add_firmware(h743_assembly SOURCES main.c)
get_target_property(fixture_config Nexus::Platform NEXUS_CONFIG_DIR)
include("${fixture_config}/selection.cmake")
set(NEXUS_CONFIG_DIR "${fixture_config}")
include("${NEXUS_SOURCE_ROOT}/cmake/platform/ControllerModes.cmake")
nexus_controller_modes(h7_fixture_uart_modes uart)
if("dma-tx" IN_LIST h7_fixture_uart_modes)
    target_compile_definitions(h743_assembly PRIVATE NX_H743_DMA_FIXTURE)
endif()
if("rx-dma-idle" IN_LIST h7_fixture_uart_modes)
    target_compile_definitions(h743_assembly PRIVATE NX_H743_RX_FIXTURE)
endif()
if("uart" IN_LIST NEXUS_SELECTED_KINDS)
    target_compile_definitions(h743_assembly PRIVATE NX_H743_IO_FIXTURE)
endif()
get_target_property(fixture_backend Nexus::Platform NEXUS_BACKEND)
if(fixture_backend STREQUAL "freertos")
    target_compile_definitions(h743_assembly PRIVATE NX_H743_FREERTOS_FIXTURE)
endif()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
if(NEXUS_SELECTED_KINDS)
    add_custom_command(TARGET h743_assembly POST_BUILD
        COMMAND "${Python3_EXECUTABLE}" -B
            "${NEXUS_SOURCE_ROOT}/tools/measurement/retention.py"
            --elf "$<TARGET_FILE:h743_assembly>"
            --bindings "${fixture_config}/bindings.c"
            --output "${CMAKE_CURRENT_BINARY_DIR}/bin/provider-retention.json"
        VERBATIM)
endif()
