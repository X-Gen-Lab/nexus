cmake_minimum_required(VERSION 3.31)
project(nexus_gd32_reference_consumer LANGUAGES C ASM)
set(NEXUS_ASSEMBLY_FILE "${CMAKE_CURRENT_LIST_DIR}/../assembly.toml" CACHE FILEPATH "")
set(NEXUS_TEST_SUPPORT ON CACHE BOOL "" FORCE)
set(NEXUS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NEXUS_BUILD_WORKLOAD OFF CACHE BOOL "" FORCE)
if(NEXUS_INSTALLED_PREFIX)
    if(NOT NEXUS_EXPECTED_SOURCE_REVISION)
        message(FATAL_ERROR "Installed reference consumption requires an exact SDK source revision")
    endif()
    find_package(Nexus CONFIG REQUIRED PATHS "${NEXUS_INSTALLED_PREFIX}/lib/cmake/Nexus" NO_DEFAULT_PATH)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/resolved-sdk.txt" "${Nexus_SOURCE_DIR}\n")
else()
    get_filename_component(NEXUS_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
    add_subdirectory("${NEXUS_SOURCE_ROOT}" nexus-sdk)
endif()
nexus_add_firmware(nexus_gd32_reference_consumer SOURCES main.c)
nexus_target_warnings(nexus_gd32_reference_consumer STRICT)
