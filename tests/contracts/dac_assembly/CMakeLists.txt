cmake_minimum_required(VERSION 3.31)
project(nexus_dac_assembly C ASM)
if(NOT IS_DIRECTORY "${NEXUS_ROOT}/io")
    message(FATAL_ERROR "Select a Nexus checkout or verified Source SDK")
endif()
set(NEXUS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NEXUS_BUILD_WORKLOAD OFF CACHE BOOL "" FORCE)
add_subdirectory("${NEXUS_ROOT}" nexus)
get_target_property(NEXUS_CONFIG_DIR Nexus::Platform NEXUS_CONFIG_DIR)
nexus_controller_modes(dac_modes dac)
if("trigger-dma" IN_LIST dac_modes OR "trigger-dma-circular" IN_LIST dac_modes)
    set(dac_consumer dma_consumer.c)
else()
    set(dac_consumer consumer.c)
endif()
nexus_add_firmware(nexus_dac_consumer SOURCES "${dac_consumer}")
nexus_target_warnings(nexus_dac_consumer STRICT)
