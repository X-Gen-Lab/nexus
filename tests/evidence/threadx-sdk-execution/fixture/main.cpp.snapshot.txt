#include "nexus/os/threadx.h"
#include "nexus_bindings.h"
#include <cstdlib>
#include <gtest/gtest.h>
#include <sched.h>

static nx_threadx_notify_t notification;
static nx_threadx_task_t producer;
static nx_threadx_permanent_task_t coordinator;
static nx_threadx_queue_t queue;
static uint8_t payload[6];
alignas(8) static uint8_t producer_stack[1024];
alignas(8) static uint8_t coordinator_stack[2048];
static nx_result_t producer_result = NX_ERROR_STATE;

/** \brief Host process reset, never a MCU shutdown or reclamation proof. */
extern "C" void nx_threadx_fail_stop(nx_result_t reason) {
  (void)reason;
  std::_Exit(66);
}

static void require(nx_result_t result) {
  if (result != NX_SUCCESS) {
    nx_threadx_fail_stop(result);
  }
}

static void produce(void *context) {
  (void)context;
  const uint8_t item[3] = {17, 23, 91};
  producer_result = nx_threadx_queue_send_until(
      &queue, item, nx_deadline_after(nx_time_now_us(), 1000000));
}

TEST(InstalledThreadX, ByteQueueAndActualJoin) {
  ASSERT_EQ(NX_SUCCESS,
            nx_threadx_task_join(&producer,
                                 nx_deadline_after(nx_time_now_us(), 1000000)));
  ASSERT_EQ(NX_SUCCESS, producer_result);
  uint8_t item[3] = {0, 0, 0};
  ASSERT_EQ(NX_SUCCESS,
            nx_threadx_queue_receive_until(&queue, item, nx_time_now_us()));
  EXPECT_EQ(17, item[0]);
  EXPECT_EQ(23, item[1]);
  EXPECT_EQ(91, item[2]);
  EXPECT_EQ(NX_SUCCESS, nx_threadx_queue_destroy(&queue));
}

TEST(InstalledThreadX, ActualMonotonicNotification) {
  nx_wait_port_t port = nx_threadx_notify_port(&notification);
  ASSERT_NE(nullptr, port.arm);
  ASSERT_NE(nullptr, port.wake);
  ASSERT_NE(nullptr, port.wait);
  const uint32_t sequence = port.arm(port.context);
  ASSERT_EQ(NX_SUCCESS, port.wake(port.context));
  EXPECT_EQ(NX_SUCCESS,
            port.wait(port.context, sequence,
                      nx_deadline_after(nx_time_now_us(), 1000000)));
  EXPECT_EQ(NX_SUCCESS, nx_threadx_notify_destroy(&notification));
}

static void run_tests(void *context) {
  (void)context;
  const int result = RUN_ALL_TESTS();
  // The coordinator is permanent. Reset the whole host process after Google
  // flushes its XML; platform clocks remain live until this process reset.
  std::_Exit(result);
}

static void startup(void *context) {
  (void)context;
  require(nx_threadx_notify_init(&notification));
  require(nx_threadx_queue_init(&queue, payload, sizeof payload, 2, 3));
  require(nx_threadx_task_start(&producer, "producer", producer_stack,
                                sizeof producer_stack, 8, produce, nullptr));
  require(nx_threadx_permanent_task_start(
      &coordinator, "coordinator", coordinator_stack, sizeof coordinator_stack,
      16, run_tests, nullptr));
}

int main(int argc, char **argv) {
  ::testing::InitGoogleTest(&argc, argv);
  // The official Linux port's scheduling model uses one allowed host CPU.
  cpu_set_t allowed;
  CPU_ZERO(&allowed);
  if (sched_getaffinity(0, sizeof allowed, &allowed) != 0) {
    return 67;
  }
  int selected = -1;
  for (int index = 0; index < CPU_SETSIZE; ++index) {
    if (CPU_ISSET(index, &allowed)) {
      selected = index;
      break;
    }
  }
  if (selected < 0) {
    return 67;
  }
  CPU_ZERO(&allowed);
  CPU_SET(selected, &allowed);
  if (sched_setaffinity(0, sizeof allowed, &allowed) != 0) {
    return 67;
  }
  const nx_platform_start_result_t result = nx_platform_start();
  if (result.primary != NX_SUCCESS || result.cleanup != NX_SUCCESS ||
      result.remaining_effects != 0) {
    return 68;
  }
  require(nx_threadx_kernel_enter(startup, nullptr));
  nx_threadx_fail_stop(NX_ERROR_STATE);
  return 69;
}
