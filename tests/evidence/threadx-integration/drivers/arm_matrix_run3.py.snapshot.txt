OUT.mkdir(exist_ok=False)

def identity(p):
 p=Path(p)
 raw=p.read_bytes()
 return {'path':str(p),'sha256':hashlib.sha256(raw).hexdigest(),'bytes':len(raw)}

import os
files = sorted(p for p in (ROOT/'os/threadx').rglob('*') if p.is_file() and '__pycache__' not in p.parts)
files += [ROOT/'tests/contracts/os_threadx_runtime/main.c', ROOT/'tests/contracts/os_threadx_runtime/CMakeLists.txt', ROOT/'tests/contracts/os_threadx_runtime/external_tick.c', ROOT/'os/CMakeLists.txt', ROOT/'dependencies/threadx.lock.json']
before=[identity(p) for p in files]
env=environment()
env['PATH']='/workspace/nexus-tools-arm/arm-gnu-toolchain-14.3.rel1-x86_64-arm-none-eabi/bin:'+env['PATH']
rows=[]
for board in ('qiming','sky','liangshan'):
 for mask in ('basepri','primask'):
  name=board+'-'+mask
  build=Path('/workspace/nexus-threadx-probe')/('arm-'+name)
  assembly=Path('/workspace/nexus-threadx-probe/arm-assemblies')/(name+'.toml')
  execute(['cmake','-S',str(ROOT/'tests/contracts/os_threadx_runtime'),'-B',str(build),'-G','Ninja','-DCMAKE_TOOLCHAIN_FILE='+str(ROOT/'cmake/toolchains/arm-gcc.cmake'),'-DNEXUS_ASSEMBLY_FILE='+str(assembly)],env,OUT/(name+'-configure.log'))
  execute(['cmake','--build',str(build),'--target','nexus_threadx_runtime','--parallel','1'],env,OUT/(name+'-build.log'))
  binary=build/'bin/nexus_threadx_runtime.elf'
  report=build/'bin/nexus_threadx_runtime.resources.json'
  data=json.loads(report.read_text())
  symbols=execute(['arm-none-eabi-nm','-S',str(binary)],env,OUT/(name+'-nm.log'))
  for symbol in ('PendSV_Handler','SysTick_Handler','__tx_PendSVHandler','_tx_timer_interrupt','nx_threadx_task_start','nx_threadx_closable_queue_close'):
   if not any(line.split()[-1]==symbol for line in symbols.splitlines()): raise ValueError('Missing retained real symbol '+symbol)
  rows.append({'board':board,'mask':mask,'scope':'actual-production-arm-link-not-execution','assembly':identity(assembly),'elf':identity(binary),'bin':identity(build/'bin/nexus_threadx_runtime.bin'),'map':identity(build/'bin/nexus_threadx_runtime.map'),'resource_report':identity(report),'resources':data,'resolved':identity(build/'nexus/generated/resolved.json'),'compile_commands':identity(build/'compile_commands.json'),'logs':[identity(OUT/(name+'-configure.log')),identity(OUT/(name+'-build.log')),identity(OUT/(name+'-nm.log'))]})
