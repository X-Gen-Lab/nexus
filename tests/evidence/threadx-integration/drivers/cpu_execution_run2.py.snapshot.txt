from pathlib import Path
import json,hashlib,sys
sys.path.insert(0,'/workspace/nexus')
from scripts.ci.tdd_gate import execute,environment
O=Path('/workspace/nexus-threadx-probe/evidence/cpu-runtime-run2');O.mkdir(exist_ok=False)
T=Path('/workspace/nexus-tools-arm/arm-gnu-toolchain-14.3.rel1-x86_64-arm-none-eabi/bin')
E=environment();E['PATH']=str(T)+':'+E['PATH']
def identity(p):
 p=Path(p);b=p.read_bytes();return {'path':str(p),'sha256':hashlib.sha256(b).hexdigest(),'bytes':len(b)}
rows=[]
for name in ['integer-basepri','integer-primask','fp-softfp']:
 build=Path('/workspace/nexus-threadx-probe')/('cpu2-'+name)
 assembly=Path('/workspace/nexus-threadx-probe/cpu-assemblies')/name/'runtime.toml'
 execute(['cmake','-S','/workspace/nexus-threadx-probe/cpu-fixture','-B',str(build),'-G','Ninja','-DCMAKE_TOOLCHAIN_FILE=/workspace/nexus/cmake/toolchains/arm-gcc.cmake','-DNEXUS_RUNTIME_ASSEMBLY='+str(assembly)],E,O/(name+'-configure.log'))
 execute(['cmake','--build',str(build),'--target','threadx_cpu','--parallel','1'],E,O/(name+'-build.log'))
 binary=build/'threadx_cpu.elf'
 symbols=execute([str(T/'arm-none-eabi-nm'),'--defined-only',str(binary)],E,O/(name+'-symbols.log'))
 for required in ['PendSV_Handler','__tx_PendSVHandler','SysTick_Handler','nx_threadx_task_start','nx_threadx_kernel_enter']:
  if not any(line.split()[-1]==required for line in symbols.splitlines()):raise ValueError(required+' discarded')
 execute([str(T/'arm-none-eabi-readelf'),'-A',str(binary)],E,O/(name+'-attributes.log'))
 rows.append({'name':name,'scope':'CPU-only-software-clock-link-not-Board-or-execution','elf':identity(binary),'assembly':identity(assembly),'facts':identity(assembly.parent/'cpu.json'),'map':identity(build/'threadx_cpu.map'),'database':identity(build/'compile_commands.json'),'resolved':identity(build/'nexus-runtime/generated/resolved-cpu.json')})
(O/'manifest.json').write_text(json.dumps({'schema':1,'qualification':'dirty-development-only','physical':'not_executed','rows':rows,'fixture':[identity('/workspace/nexus-threadx-probe/cpu-fixture/main.c'),identity('/workspace/nexus-threadx-probe/cpu-fixture/CMakeLists.txt')],'commands':[identity(p) for p in sorted(O.glob('*.command.json'))]},indent=2)+'\n')
print('actual CPU links',len(rows))
