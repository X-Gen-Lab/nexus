function(nexus_service_test name)
    add_executable(${name} ${ARGN})
    # C assertions in these standalone tests remain active in Release builds.
    if(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "service;contract" TIMEOUT 60)
endfunction()

    nexus_service_test(crypto_unconfigured_tests test_crypto_disabled.c)
    target_link_libraries(crypto_unconfigured_tests PRIVATE nexus_security)
endif()
