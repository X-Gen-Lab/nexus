cmake_minimum_required(VERSION 3.31)
project(NexusTlsHostPeer C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT IS_DIRECTORY "${NEXUS_MBEDTLS_SOURCE}" OR
   NOT IS_DIRECTORY "${NEXUS_TLS_PROFILE_ROOT}")
    message(FATAL_ERROR "Host peer requires the already admitted TLS/profile sources")
endif()
set(vendor_source "${NEXUS_MBEDTLS_SOURCE}")
set(ENABLE_PROGRAMS OFF CACHE BOOL "No programs")
set(ENABLE_TESTING OFF CACHE BOOL "No supplier tests")
set(GEN_FILES OFF CACHE BOOL "Reviewed generated materials")
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "Static TLS")
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "Static TLS")
set(MBEDTLS_CONFIG_FILE "${vendor_source}/configs/config-suite-b.h" CACHE FILEPATH "TLS")
set(TF_PSA_CRYPTO_CONFIG_FILE "${vendor_source}/configs/crypto-config-suite-b.h" CACHE FILEPATH "Crypto")
set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/profile/tls-user.h" CACHE FILEPATH "Peer-only TLS")
set(TF_PSA_CRYPTO_USER_CONFIG_FILE "${NEXUS_TLS_PROFILE_ROOT}/crypto-user.h" CACHE FILEPATH "Explicit entropy and arena")
add_subdirectory("${vendor_source}" vendor)
foreach(target builtin everest extras mbedtls mbedx509 p256-m platform pqcp tfpsacrypto utilities)
    if(NOT TARGET "${target}")
        message(FATAL_ERROR "Reviewed upstream target absent: ${target}")
    endif()
    set_target_properties("${target}" PROPERTIES
        C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
endforeach()
add_executable(tls_host_peer server.c)
target_link_libraries(tls_host_peer PRIVATE MbedTLS::mbedtls)
target_compile_options(tls_host_peer PRIVATE -Os -Wall -Wextra -Werror)
